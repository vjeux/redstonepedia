# turns on, and the observer fires once as it lands, into the cell behind it (the output, a 2-tick
# pulse); pulled back down when the input turns off, it fires under the output, into nothing.
# Copies stand side by side every 2 blocks. Detail: inputs of 6 ticks or more.

port output: 0 2 -1
